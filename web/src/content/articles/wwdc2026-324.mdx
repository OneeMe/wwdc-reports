## Highlight

> Core AI 让"把 PyTorch 模型搬到 iPhone 上跑"这件事变得像写几行 Python 脚本一样简单——从模型转换、优化到 Swift 端推理，整条链路 Apple 都帮你铺好了，而且模型在设备本地运行，没有服务器成本，没有隐私顾虑。

## 核心内容

### 1. 这场演讲解决什么问题

过去想在 Apple 设备上跑自定义 AI 模型，开发者要面对一系列复杂选择：Core ML 的转换工具链、Metal Performance Shaders 的手写 kernel、或者直接把模型放在服务器端。每种方案都有代价——要么转换失败，要么开发成本高，要么有网络延迟和隐私风险。

Core AI 是 Apple 今年推出的全新框架，定位是"在 Apple Silicon 上跑自定义模型的最佳方式"。它覆盖模型部署的完整生命周期：Python 端的转换和优化、Swift 端的推理 API、以及 Xcode 集成的调试和性能分析工具。

### 2. PyTorch 模型转换：从 .pt 到 .aimodel

Core AI 的 Python 工具链 `coreai-torch` 可以直接消费 PyTorch 的 `torch.export` 输出，不需要中间格式。

```python
import torch
import coreai_torch

# 加载训练好的模型和示例输入
pt_model = SnakeTransformer().load_checkpoint("snake.pt")
example = torch.randn(1, 5, 16)

# 定义动态维度（序列长度可变）
seq_len = torch.export.Dim("seq_len", min=1, max=256)
exported = torch.export.export(
    pt_model, args=(example,),
    dynamic_shapes={"features": {1: seq_len}},
)
exported = exported.run_decompositions(coreai_torch.get_decomp_table())

# 转换 Torch 计算图 → Core AI 计算图
ai_program = coreai_torch.TorchConverter().add_exported_program(
    exported, input_names=["features"], output_names=["logits"],
).to_coreai()

# 保存为 .aimodel 资源文件
ai_program.save_asset("SnakeTransformer.aimodel")
```

转换完成后，建议验证数值一致性：

```python
import torch
import numpy as np
from coreai.runtime import AIModel, NDArray

# 加载两个模型
pt_model = SnakeTransformer().load_checkpoint("snake.pt")
ai_model = await AIModel.load("SnakeTransformer.aimodel")
function = ai_model.load_function("main")

# 构造输入
features = np.array([extract_features(game) for _ in range(10)],
                    dtype=np.float32)[np.newaxis]

# PyTorch 推理
with torch.no_grad():
    pytorch_logits = pt_model(torch.from_numpy(features)).numpy()[0, -1]

# Core AI 推理
result = await function({"features": NDArray(data=features)})
coreai_logits = result["logits"].numpy()[0, -1]

# 验证
max_diff = np.max(np.abs(pytorch_logits - coreai_logits))
assert max_diff < 0.01
```

关键点：

- `torch.export.export()` 捕获完整的计算图，包括权重、操作和形状信息
- `dynamic_shapes` 让模型支持可变输入尺寸（如不同长度的序列）
- `run_decompositions()` 把 PyTorch 操作分解为 Core AI 支持的原子操作
- 数值验证是必要步骤，确保转换没有引入精度损失

### 3. Swift 端推理：AIModel + InferenceFunction

转换好的 `.aimodel` 文件在 Swift 中的使用非常直接：

```swift
import CoreAI

// 加载模型
let model = try await AIModel(contentsOf: modelURL)

// 加载主推理函数
let mainFunction: InferenceFunction = try model.loadFunction(named: "main")!

// 构造输入数据
let inputNDArray: NDArray = nextInput()

// 运行推理
var outputs = try await mainFunction.run(inputs: ["input": inputNDArray])

guard let outputNDArray = outputs.remove("output")?.ndArray else {
    // 处理异常输出
}
```

用一个具体的游戏 AI 例子来看完整流程：

```swift
struct ModelPlayer {
    let nextActionFunction: InferenceFunction

    init(modelURL: URL) async throws {
        let model = try await AIModel(contentsOf: modelURL)
        self.nextActionFunction = try model.loadFunction(named: "main")!
    }
}

extension ModelPlayer: SnakePlayer {
    mutating func chooseAction(game: SnakeGame) async throws -> Direction {
        // 创建 NDArray 并写入游戏特征
        var inputFeatures = NDArray(
            shape: [game.stepCount, hiddenDim],
            scalarType: .float32
        )
        writeFeatures(of: game, into: inputFeatures.mutableView())

        // 运行推理
        var outputs = try await nextActionFunction.run(
            inputs: ["features": inputFeatures]
        )
        guard let logits = outputs.remove("logits")?.ndArray else {
            throw ModelError.missingOutput
        }

        return predictedDirection(from: logits.view())
    }
}
```

关键点：

- `AIModel` 是模型的容器，`InferenceFunction` 是具体的推理入口
- `NDArray` 是 Core AI 的核心数据类型，支持多维数组和多种标量类型
- `mutableView()` 和 `view()` 分别用于读写和只读访问，保证内存安全
- 输入输出通过字典按名称映射，和 PyTorch 端的 `input_names`/`output_names` 对应

### 4. 状态管理：KV Cache 优化

对于 Transformer 这类自回归模型，每次推理都需要重新计算之前 token 的 key 和 value，浪费严重。Core AI 支持状态ful 推理，让模型在多次调用之间缓存中间结果。

PyTorch 端改造模型，把 KV cache 注册为 buffer：

```python
class SnakeTransformerStateful(nn.Module):
    def __init__(self, ...):
        super().__init__()
        self.register_buffer(
            "k_cache", torch.zeros(N_LAYERS, 1, MAX_SEQ_LEN, D_MODEL))
        self.register_buffer(
            "v_cache", torch.zeros(N_LAYERS, 1, MAX_SEQ_LEN, D_MODEL))

    def forward(self, features, position_ids):
        new_k, new_v = [], []
        for i, block in enumerate(self.blocks):
            k_prev = self.k_cache[i]
            v_prev = self.v_cache[i]
            # ... 计算并更新 cache
            new_k.append(k_updated)
            new_v.append(v_updated)

        self.k_cache.copy_(torch.stack(new_k))
        self.v_cache.copy_(torch.stack(new_v))
        return self.action_head(self.ln_final(x))
```

转换时声明状态名称：

```python
ai_program = coreai_torch.TorchConverter().add_exported_program(
    exported,
    input_names=["features", "position_ids"],
    state_names=["keyCache", "valueCache"],  // 声明状态
    output_names=["logits"],
).to_coreai()
```

Swift 端在多次推理之间保持状态：

```swift
struct ModelPlayer {
    let nextActionFunction: InferenceFunction
    var keyCache: NDArray
    var valueCache: NDArray

    init(modelURL: URL) async throws {
        let model = try await AIModel(contentsOf: modelURL)
        self.nextActionFunction = try model.loadFunction(named: "main")!
        self.keyCache = NDArray(shape: [layers, maxContext, hiddenDim],
                                scalarType: .float32)
        self.valueCache = NDArray(shape: [layers, maxContext, hiddenDim],
                                  scalarType: .float32)
    }

    mutating func chooseAction(game: SnakeGame) async throws -> Direction {
        // ... 准备 inputFeatures ...

        var stateViews = InferenceFunction.MutableViews()
        stateViews.insert(&keyCache, for: "keyCache")
        stateViews.insert(&valueCache, for: "valueCache")

        var outputs = try await nextActionFunction.run(
            inputs: ["features": inputFeatures],
            states: stateViews  // 传入状态
        )
        // ...
    }
}
```

关键点：

- `register_buffer` 让 PyTorch 把 KV cache 当作模型状态而非可训练参数
- `state_names` 在转换时把 buffer 映射为 Core AI 的状态接口
- Swift 端用 `MutableViews` 管理状态，多次推理之间状态自动保持
- 这让自回归模型的每次后续推理从 O(n) 降到 O(1)

### 5. 模型预热与缓存

首次加载模型时，Core AI 会进行设备特化（specialization）——针对具体设备的 GPU/ANE 架构编译优化。这个过程可能较慢，建议提前触发：

```swift
// 检查缓存中是否已有特化后的模型
let cache = AIModelCache.default
guard let model = try cache.model(for: modelURL, options: .default) else {
    // 没有缓存，提示用户等待
    Task { @MainActor in
        informUser("Preparing AI features. This may take a while...")
    }
}

// 显式请求特化（在后台提前完成）
try await AIModel.specialize(contentsOf: modelURL)
```

## 最佳实践

- **先验证再集成**：转换后务必做数值验证，确保 PyTorch 和 Core AI 的输出差异在可接受范围
- **用动态形状处理变长输入**：序列长度、图像尺寸等可变维度用 `torch.export.Dim` 声明，避免为每种尺寸单独转换
- **KV Cache 是自回归模型的必选项**：没有状态管理，Transformer 的推理速度会随序列长度线性下降
- **提前特化模型**：在 App 启动或用户进入相关功能页时调用 `specialize()`，避免首次推理的冷启动延迟
- **利用 Xcode 的 Core AI Debugger**：可以追踪 tensor 值直接回溯到原始 Python 源码，调试转换问题
